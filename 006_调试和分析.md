## 打印问题

直接使用print函数打印要观察的参数。

```
import torch
import torch.nn as nn


def debug_print(name, tensor):
    print(f"{name}: shape={tensor.shape}, dtype={tensor.dtype}, "
          f"device={tensor.device}, "
          f"min={tensor.min().item():.4f}, max={tensor.max().item():.4f}, "
          f"mean={tensor.mean().item():.4f}, "
          f"has_nan={tensor.isnan().any().item()}")


# ==================== 模型定义 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(10, 32)
        self.fc2 = nn.Linear(32, 1)

    def forward(self, x):
        x = self.fc1(x)
        debug_print("fc1_out", x)          # 追踪中间激活
        x = torch.relu(x)
        x = self.fc2(x)
        debug_print("fc2_out", x)
        return x


# ==================== 训练流程 ====================
model = Net()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
criterion = nn.MSELoss()

for step in range(5):
    x = torch.randn(8, 10)
    y = torch.randn(8, 1)

    debug_print("input_x", x)              # 输入
    debug_print("target_y", y)             # 标签

    out = model(x)                         # forward（内部已有 debug）
    debug_print("output", out)             # 输出

    loss = criterion(out, y)
    debug_print("loss", loss)              # loss

    optimizer.zero_grad()
    loss.backward()

    # 反向传播后检查梯度
    debug_print("fc1.weight.grad", model.fc1.weight.grad)
    debug_print("fc2.weight.grad", model.fc2.weight.grad)
    debug_print("fc1.bias.grad", model.fc1.bias.grad)

    optimizer.step()

    # 更新后检查参数
    debug_print("fc1.weight", model.fc1.weight)
    debug_print("fc2.weight", model.fc2.weight)
    print("-" * 60)
```

## PDB调试

breakpoint() 进入训练循环,并进行互动检查.

```

import torch
import torch.nn as nn


def training_step(model, batch, criterion, optimizer):
    inputs, labels = batch
    outputs = model(inputs)
    loss = criterion(outputs, labels)

    if loss.item() > 100 or torch.isnan(loss):
        breakpoint()

    loss.backward()
    optimizer.step()


# ==================== 模型 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(10, 32)
        self.fc2 = nn.Linear(32, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        return self.fc2(x)


model = Net()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
criterion = nn.MSELoss()


# ==================== 场景 1：标签尺度极大 ====================
# 标签放大 1000 倍，MSE loss 轻松 > 100
print(">>> 场景 1：labels 放大 1000 倍")
inputs = torch.randn(8, 10)
labels = torch.randn(8, 1) * 1000.0   # 关键：让 loss 爆炸

optimizer.zero_grad()
training_step(model, (inputs, labels), criterion, optimizer)


# ==================== 场景 2：输入含 NaN ====================
# 输入含 NaN -> outputs 含 NaN -> loss 为 NaN
print(">>> 场景 2：inputs 含 NaN")
inputs = torch.randn(8, 10)
inputs[0, 0] = float("nan")           # 关键：注入 NaN
labels = torch.randn(8, 1)

optimizer.zero_grad()
training_step(model, (inputs, labels), criterion, optimizer)


# ==================== 场景 3：权重大 -> 输出爆炸 ====================
# 手动把权重放大，输出随之爆炸
print(">>> 场景 3：权重放大 100 倍")
with torch.no_grad():
    model.fc2.weight.mul_(100.0)
    model.fc2.bias.mul_(100.0)

inputs = torch.randn(8, 10)
labels = torch.randn(8, 1)

optimizer.zero_grad()
training_step(model, (inputs, labels), criterion, optimizer)
```

## 增加日志记录

```
import logging
import torch
import torch.nn as nn

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[
        logging.FileHandler("training.log"),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)


def training_step(model, batch, criterion, optimizer):
    inputs, labels = batch
    outputs = model(inputs)
    loss = criterion(outputs, labels)

    loss.backward()
    optimizer.step()


# ==================== 模型 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(10, 32)
        self.fc2 = nn.Linear(32, 1)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        return self.fc2(x)


# ==================== 训练循环 ====================
lr = 1e-3
batch_size = 8

model = Net()
optimizer = torch.optim.Adam(model.parameters(), lr=lr)
criterion = nn.MSELoss()

logger.info("Starting training: lr=%.4f, batch_size=%d", lr, batch_size)

for step in range(10):
    # 场景：让 loss 在某个 step 爆炸，触发 warning / error
    inputs = torch.randn(batch_size, 10)
    labels = torch.randn(batch_size, 1)

    if step == 3:
        # 让 loss spike（label 放大）
        labels = labels * 1000.0
        logger.info("Step %d: injected label spike (x1000)", step)

    if step == 6:
        # 让 loss 变 NaN（输入注入 NaN）
        inputs[0, 0] = float("nan")
        logger.info("Step %d: injected NaN into input", step)

    optimizer.zero_grad()
    outputs = model(inputs)
    loss = criterion(outputs, labels)

    # ---- 日志：正常记录 ----
    logger.info("Step %d: loss=%.4f", step, loss.item())

    # ---- 日志：loss 爆炸 ----
    if loss.item() > 100:
        logger.warning("Loss spike detected: %.4f at step %d", loss.item(), step)

    # ---- 日志：NaN loss，停止 ----
    if torch.isnan(loss):
        logger.error("NaN loss at step %d, stopping", step)
        break

    # ---- 调用 training_step ----
    training_step(model, (inputs, labels), criterion, optimizer)
```

## 度量耗时

```
import time
import torch
import torch.nn as nn


class Timer:
    def __init__(self, name=""):
        self.name = name

    def __enter__(self):
        self.start = time.perf_counter()
        return self

    def __exit__(self, *args):
        elapsed = time.perf_counter() - self.start
        print(f"[{self.name}] {elapsed:.4f}s")


# ==================== 模型 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(64, 256)
        self.fc2 = nn.Linear(256, 10)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        return self.fc2(x)


# ==================== 模拟数据集 ====================
class FakeDataset(torch.utils.data.Dataset):
    def __init__(self, n=1000, dim=64):
        self.n = n
        self.dim = dim

    def __len__(self):
        return self.n

    def __getitem__(self, idx):
        # 模拟慢速数据加载（磁盘/网络 IO）
        time.sleep(1)
        return torch.randn(self.dim), torch.randint(0, 10, ())


dataset = FakeDataset()
dataloader = torch.utils.data.DataLoader(dataset, batch_size=32)
dataloader_iter = iter(dataloader)

model = Net()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
criterion = nn.CrossEntropyLoss()


# ==================== 训练循环 ====================
for step in range(5):
    print(f"===== step {step} =====")

    # ---- 1. 数据加载 ----
    with Timer("data loading"):
        batch = next(dataloader_iter)

    # ---- 2. 前向传播 ----
    with Timer("forward pass"):
        outputs = model(batch[0])

    # ---- 3. loss 计算 ----
    with Timer("loss compute"):
        loss = criterion(outputs, batch[1])

    # ---- 4. 反向传播 ----
    optimizer.zero_grad()
    with Timer("backward pass"):
        loss.backward()

    # ---- 5. 优化器更新 ----
    with Timer("optimizer step"):
        optimizer.step()

    print(f"loss={loss.item():.4f}")
```

## torch.profiler

torch.profiler能看到 kernel 级别的 GPU 耗时。

```
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.profiler import profile, ProfilerActivity


# ==================== 模型 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(64, 256)
        self.fc2 = nn.Linear(256, 10)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        return self.fc2(x)


# ==================== training_step ====================
def training_step(model, batch, criterion, optimizer):
    inputs, labels = batch
    outputs = model(inputs)
    loss = criterion(outputs, labels)
    loss.backward()
    optimizer.step()
    return loss


# ==================== 运行 + profile ====================
def main():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    model = Net().to(device)
    optimizer = torch.optim.SGD(model.parameters(), lr=1e-3)
    criterion = nn.CrossEntropyLoss()

    batch = (
        torch.randn(32, 64).to(device),
        torch.randint(0, 10, (32,)).to(device),
    )

    # ---- 预热：让 CUDA/cuDNN 初始化完成，避免污染 profile ----
    for _ in range(5):
        optimizer.zero_grad()
        training_step(model, batch, criterion, optimizer)

    # ---- 正式 profile ----
    optimizer.zero_grad()
    with profile(activities=[ProfilerActivity.CPU, ProfilerActivity.CUDA]) as prof:
        training_step(model, batch, criterion, optimizer)

    # 按 CUDA 耗时排序
    print(prof.key_averages().table(sort_by="cuda_time_total", row_limit=10))


if __name__ == "__main__":
    main()
```

## tensorboard

pip install tensorboard

```
import torch
import torch.nn as nn
from torch.utils.tensorboard import SummaryWriter


# ==================== 模型 ====================
class Net(nn.Module):
    def __init__(self):
        super().__init__()
        self.fc1 = nn.Linear(64, 256)
        self.fc2 = nn.Linear(256, 10)

    def forward(self, x):
        x = torch.relu(self.fc1(x))
        return self.fc2(x)


# ==================== train_step ====================
def train_step(model, batch, criterion, optimizer):
    inputs, labels = batch
    outputs = model(inputs)
    loss = criterion(outputs, labels)

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    return loss


# ==================== 主流程 ====================
def main():
    # ---- 配置 ----
    num_steps = 300
    batch_size = 32
    lr = 1e-3

    device = "cuda" if torch.cuda.is_available() else "cpu"

    model = Net().to(device)
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=lr)

    # ---- TensorBoard ----
    writer = SummaryWriter("runs/experiment_1")

    for step in range(num_steps):
        # 模拟一个 batch
        batch = (
            torch.randn(batch_size, 64).to(device),
            torch.randint(0, 10, (batch_size,)).to(device),
        )

        loss = train_step(model, batch, criterion, optimizer)

        # 标量日志
        writer.add_scalar("loss/train", loss.item(), step)
        writer.add_scalar("lr", optimizer.param_groups[0]["lr"], step)

        # 每 100 步记录权重和梯度的直方图
        if step % 100 == 0:
            for name, param in model.named_parameters():
                writer.add_histogram(f"weights/{name}", param, step)
                if param.grad is not None:
                    writer.add_histogram(f"grads/{name}", param.grad, step)

        if step % 50 == 0:
            print(f"step {step}: loss={loss.item():.4f}")

    writer.close()
    print("done. run: tensorboard --logdir=runs")


if __name__ == "__main__":
    main()
```
